Upon review of the WebKit implementation I found that a significant number of issues could be traced back to this algorithm.

Might be better to already hint at this issue in the sample algorithm and to give a shape that better translates to implementations that use floats?

```js
function Lab_to_XYZ(Lab) {
	// Convert Lab to D50-adapted XYZ
	// http://www.brucelindbloom.com/index.html?Eqn_Lab_to_XYZ.html
	var κ = 24389/27;   // 29^3/3^3
	var ε = 216/24389;  // 6^3/29^3
	var f = [];

	// Keep L out of L+16 for near-black precision
	var lightness116 = Lab[0]/116;
	var a500 = Lab[1]/500;
	var b200 = Lab[2]/200;

	// compute f, starting with the luminance-related term
	f[1] = lightness116 + (16/116);
	f[0] = f[1] + a500;
	f[2] = f[1] - b200;

	// compute xyz
	var xyz = [
		Math.pow(f[0],3) > ε ?   Math.pow(f[0],3)            : (lightness116 + a500) * (116/κ),
		Lab[0] > κ * ε ?         Math.pow((Lab[0]+16)/116,3) : Lab[0]/κ,
		Math.pow(f[2],3) > ε ?   Math.pow(f[2],3)            : (lightness116 - b200) * (116/κ)
	];

	// Compute XYZ by scaling xyz by reference white
	return xyz.map((value, i) => value * D50[i]);
}
```
